    <script>

        const formatIDR = (n) => new Intl.NumberFormat('id-ID', { style: 'currency', currency: 'IDR', minimumFractionDigits: 0 }).format(n);

        

        let portfolio = [

            { id: 'VOO', name: 'S&P 500', target: 0.30, color: '#4f46e5' },

            { id: 'QQQ', name: 'Nasdaq', target: 0.20, color: '#6366f1' },

            { id: 'SCHD', name: 'Dividen', target: 0.15, color: '#818cf8' },

            { id: 'AAPL', name: 'Apple', target: 0.10, color: '#1e293b' },

            { id: 'MSFT', name: 'Microsoft', target: 0.10, color: '#334155' },

            { id: 'GOOGL', name: 'Google', target: 0.05, color: '#475569' },

            { id: 'AMZN', name: 'Amazon', target: 0.05, color: '#64748b' },

            { id: 'BRK.B', name: 'Berkshire', target: 0.05, color: '#94a3b8' }

        ];


        let chart;

        let currentCommands = [];


        // Inisialisasi Chart

        window.onload = () => {

            const ctx = document.getElementById('liveChart').getContext('2d');

            chart = new Chart(ctx, {

                type: 'bar',

                data: {

                    labels: portfolio.map(a => a.id),

                    datasets: [

                        { label: 'Kondisi Saat Ini (%)', data: portfolio.map(() => 0), backgroundColor: '#6366f1', borderRadius: 8 },

                        { label: 'Target Ideal (%)', data: portfolio.map(a => a.target * 100), backgroundColor: '#f1f5f9', borderRadius: 8 }

                    ]

                },

                options: { responsive: true, maintainAspectRatio: false, plugins: { legend: { display: false } }, scales: { y: { beginAtZero: true, max: 45 } } }

            });

            renderJournal();

            checkMarket();

        };


        async function updateData() {

            const key = document.getElementById('apiKey').value;

            const capital = parseFloat(document.getElementById('capitalInput').value);

            if (!key) { alert("Silakan masukkan Finnhub API Key Anda."); return; }


            try {

                const fxRes = await fetch('https://api.exchangerate-api.com/v4/latest/USD');

                const fxData = await fxRes.json();

                const rate = fxData.rates.IDR;

                document.getElementById('fxRate').innerText = formatIDR(rate);


                let totalValue = 0;

                // Fetching prices

                const results = await Promise.all(portfolio.map(a => 

                    fetch(`https://finnhub.io/api/v1/quote?symbol=${a.id}&token=${key}`).then(r => r.json())

                ));

                

                portfolio.forEach((asset, i) => {

                    const priceInIDR = results[i].c * rate;

                    // Simulasi drift: Dalam app asli, user memasukkan unit yang mereka punya. 

                    // Di sini kita asumsikan porsi awal adalah target dan harga berubah secara real.

                    asset.currentVal = (capital * asset.target) * (results[i].c / (results[i].pc || results[i].c));

                    totalValue += asset.currentVal;

                });


                renderUI(totalValue, capital);

            } catch (e) { alert("Error mengambil data. Cek API Key Anda."); }

        }


        function renderUI(total, initial) {

            document.getElementById('currentTotal').innerText = formatIDR(total);

            const diff = total - initial;

            document.getElementById('totalChange').innerHTML = `<span class="${diff >= 0 ? 'text-emerald-600' : 'text-rose-600'}">${diff >= 0 ? '▲' : '▼'} ${formatIDR(Math.abs(diff))} (${((diff/initial)*100).toFixed(2)}%)</span>`;


            chart.data.datasets[0].data = portfolio.map(a => (a.currentVal / total) * 100);

            chart.update();


            const rebList = document.getElementById('rebalanceList');

            const assetList = document.getElementById('assetList');

            rebList.innerHTML = '';

            assetList.innerHTML = '';

            currentCommands = [];


            portfolio.forEach(a => {

                const weight = (a.currentVal / total);

                const drift = ((weight - a.target) / a.target) * 100;


                assetList.innerHTML += `

                    <div class="flex justify-between items-center text-sm">

                        <span class="font-bold text-slate-700">${a.id}</span>

                        <div class="text-right">

                            <p class="font-bold">${(weight*100).toFixed(1)}%</p>

                            <p class="text-[10px] ${Math.abs(drift) > 2 ? 'text-orange-500 font-bold' : 'text-slate-400'}">${drift > 0 ? '+' : ''}${drift.toFixed(1)}% drift</p>

                        </div>

                    </div>`;


                if (Math.abs(drift) > 2.5) {

                    const amount = Math.abs(a.currentVal - (total * a.target));

                    const isSell = drift > 0;

                    currentCommands.push({ asset: a.id, action: isSell ? 'JUAL' : 'BELI', amount: amount });

                    

                    rebList.innerHTML += `

                        <div class="bg-white/10 p-5 rounded-2xl border border-white/10 flex justify-between items-center">

                            <div>

                                <p class="text-[10px] font-bold opacity-60 uppercase">${isSell ? 'Take Profit' : 'Serok Bawah'}</p>

                                <p class="font-bold text-lg">${isSell ? 'Jual' : 'Beli'} ${a.id}</p>

                            </div>

                            <p class="text-right font-black text-indigo-300">${formatIDR(amount)}</p>

                        </div>`;

                }

            });


            if (currentCommands.length === 0) {

                rebList.innerHTML = '<p class="text-center py-12 opacity-40 italic">Portofolio Anda sudah sangat seimbang. Nikmati kopi Anda! ☕</p>';

                document.getElementById('logActionBtn').style.display = 'none';

            } else {

                document.getElementById('logActionBtn').style.display = 'block';

            }

        }


        function logTransactions() {

            let journal = JSON.parse(localStorage.getItem('gotrade_journal') || '[]');

            const now = new Date().toLocaleString('id-ID');

            

            currentCommands.forEach(cmd => {

                journal.unshift({ date: now, action: cmd.action, asset: cmd.asset, amount: cmd.amount });

            });

            

            localStorage.setItem('gotrade_journal', JSON.stringify(journal));

            renderJournal();

            alert("Berhasil mencatat rebalancing ke jurnal!");

        }


        function renderJournal() {

            const journal = JSON.parse(localStorage.getItem('gotrade_journal') || '[]');

            const body = document.getElementById('journalBody');

            const empty = document.getElementById('emptyJournal');

            

            body.innerHTML = '';

            if (journal.length === 0) {

                empty.style.display = 'block';

                return;

            }

            

            empty.style.display = 'none';

            journal.forEach(item => {

                body.innerHTML += `

                    <tr class="border-b border-slate-50 hover:bg-slate-50/50 transition-colors">

                        <td class="px-6 py-4 text-slate-500 text-[11px]">${item.date}</td>

                        <td class="px-6 py-4"><span class="px-2 py-1 rounded-md text-[10px] font-bold ${item.action === 'BELI' ? 'bg-emerald-100 text-emerald-700' : 'bg-rose-100 text-rose-700'}">${item.action}</span></td>

                        <td class="px-6 py-4 font-bold text-slate-700">${item.asset}</td>

                        <td class="px-6 py-4 text-right font-bold">${formatIDR(item.amount)}</td>

                    </tr>`;

            });

        }


        function clearJournal() {

            if (confirm("Hapus semua catatan histori?")) {

                localStorage.removeItem('gotrade_journal');

                renderJournal();

            }

        }


        function checkMarket() {

            const h = new Date().getUTCHours();

            const d = new Date().getUTCDay();

            const isOpen = (d >= 1 && d <= 5) && (h >= 13 && h <= 21); // Simplified US Market Hours

            const el = document.getElementById('marketStatus');

            el.innerHTML = isOpen ? `<span class="flex items-center gap-2"><span class="w-2 h-2 rounded-full bg-emerald-500 animate-pulse"></span> Market Open</span>` : `<span class="flex items-center gap-2"><span class="w-2 h-2 rounded-full bg-slate-300"></span> Market Closed</span>`;

        }


        document.getElementById('updateBtn').onclick = updateData;

        setInterval(checkMarket, 60000);
